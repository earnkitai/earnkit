# v2.3 Token Launch at Submission — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every builder who submits through EarnKit gets a token by default. EarnKit proposes the name, ticker and logo; the builder can edit it or say "no token". A launch happens only if the repo is real and the live link loads.

**Architecture:**
- `prepare_submission` returns a proposed token: name, ticker, logo source, earnings line and terms.
- `record_submission` records the entry exactly as today. It then calls the new `tokenAfterSubmission`, which applies the program's `token_policy`, runs the validity gate, and launches through the existing tokens.fun deploy route.
- The submission is never rolled back because of a token problem; the reply always includes `token: { status, … }`.
- Outside the app:
  - tokens.fun accepts a second API key that skips its per-IP limits;
  - the mainnet contract deploy is locked to the company treasury.

**Tech Stack:** Next.js 16 (app router, `next/og`), postgres.js, zod v4, node:test via `tsx --test`, Foundry (contracts), vitest (tokens.fun).

**Spec:** `docs/design-docs-and-implementation-plans/v2.3-token-at-submission.md`

```ts
import { timingSafeCompare } from "./timingSafeCompare";

/**
 * Which server key a request carries: the shared deploy key ("main"), EarnKit's own key
 * ("earnkit"), or neither. EarnKit's launches all come from one server IP, so its key skips the
 * per-IP limits in /api/v1/deploy; the per-creator-wallet daily cap still applies to it.
 * Keys are read on each call; an unset key never matches.
 */
export function deployCaller(authHeader: string | null): "main" | "earnkit" | null {
  if (!authHeader?.startsWith("Bearer ")) return null;
  const bearer = authHeader.slice(7);
  if (!bearer) return null;
  const main = process.env.DEPLOY_API_KEY ?? "";
  const earnkit = process.env.EARNKIT_DEPLOY_API_KEY ?? "";
  if (main && timingSafeCompare(bearer, main)) return "main";
  if (earnkit && timingSafeCompare(bearer, earnkit)) return "earnkit";
  return null;
}
```

/** Same site for a redirect: the same host, ignoring one leading "www.". Pure; exported for tests. */
export const sameSite = (a: string, b: string) =>
  a.replace(/^www\./, '') === b.replace(/^www\./, '');

/**
 * Whether a builder's live link loads: any 2xx, the body never read. Follows one redirect within
 * the same site (apex↔www, a path change); refuses a redirect elsewhere or a second redirect.
 */
export async function safeProbe(
  url: string,
  deps: { resolve?: Resolver; transport?: Transport; timeoutMs?: number } = {},
): Promise<void> {
  try {
    await safeFetchText(url, { ...deps, probe: true });
  } catch (err) {
    if (!(err instanceof RedirectError) || !err.location) throw err;
    const next = new URL(err.location, url);
    if (!sameSite(next.hostname, new URL(url).hostname))
      throw new SafeFetchError(`redirects to another site (${next.hostname})`);
    await safeFetchText(next.href, { ...deps, probe: true });
  }
}

import { githubRepo } from './ownership';
import { repoByName, repoHasCommits } from './match-projects';
import type { TokenPolicy } from './program-file';
import { safeProbe } from './safe-fetch';
import { BASE_MAINNET, screenWallet } from './sanctions';

// (GateInput, GateDeps, GateResult types as in Interfaces above)

const defaults: Required<GateDeps> = {
  repo: async (owner, repo) => {
    const r = await repoByName(`${owner}/${repo}`);
    return { fork: r.fork, hasCommits: await repoHasCommits(`${owner}/${repo}`) };
  },
  probe: (url) => safeProbe(url),
  screen: (wallet) => screenWallet(wallet, BASE_MAINNET, "This wallet can't receive token fees on EarnKit."),
};

const msg = (err: unknown) => (err instanceof Error ? err.message : String(err));
const RETRY = " Fix it and say 'launch the token'.";

/**
 * Whether a recorded project may get a token (v2.3 §3): the program allows tokens, the public
 * GitHub repo exists with commits and isn't a fork, the live link loads, and the creator wallet
 * isn't sanctioned. Stops at the first failure; never cached.
 */
export async function tokenGate(input: GateInput, deps: GateDeps = {}): Promise<GateResult> {
  const d = { ...defaults, ...deps };
  if (input.policy === 'off') return { ok: false, reason: "This program doesn't allow tokens for entries." };
  const gh = input.repoUrl ? githubRepo(input.repoUrl) : null;
  if (!gh) return { ok: false, reason: 'A token needs the project\'s public GitHub repo (repo_url).' + RETRY };
  try {
    const facts = await d.repo(gh.owner, gh.repo);
    if (facts.fork) return { ok: false, reason: "The repo is a fork; tokens are for original projects." };
    if (!facts.hasCommits) return { ok: false, reason: 'The repo has no commits yet.' + RETRY };
  } catch (err) {
    return /not found/.test(msg(err))
      ? { ok: false, reason: `The repo ${gh.owner}/${gh.repo} isn't public on GitHub.` + RETRY }
      : { ok: false, reason: "EarnKit couldn't check the repo on GitHub right now. Say 'launch the token' to retry." };
  }
  if (!input.liveUrl) return { ok: false, reason: 'There is no live link. A token needs a project that runs: add live_url and say \'launch the token\'.' };
  try {
    await d.probe(input.liveUrl);
  } catch (err) {
    return { ok: false, reason: `The live link didn't load (${msg(err)}).` + RETRY };
  }
  try {
    await d.screen(input.wallet);
  } catch (err) {
    return { ok: false, reason: `creator wallet: ${msg(err)}` };
  }
  return { ok: true };
}

import { reservedBrands } from './sponsor-identity';
import { UntrustedTextError, cleanText } from './untrusted-text';

const TOKENSFUN_URL = (process.env.TOKENSFUN_API_URL ?? 'https://tokens.fun').replace(/\/$/, '');

/** Tickers that read as a brand's or a major asset's own (v2.3 §2). */
export const TICKER_RESERVED = [
  'BASE', 'CBBTC', 'CBETH', 'USDC', 'USDT', 'EURC', 'ETH', 'WETH', 'BTC', 'WBTC',
  'MINI', 'EARN', 'EARNKIT', 'COIN', 'COINBASE', 'CB', 'ARC', 'MONAD', 'MON',
  'HOOD', 'ROBINHOOD', 'TEMPO', 'CIRCLE', 'CLANKER', 'TOKENS',
] as const;

/** Why `name` can't be a token name, or null. Pure; exported for tests. */
export function tokenNameProblem(name: string): string | null {
  try {
    const clean = cleanText(name, { max: 50, singleLine: true, field: 'name' });
    if (reservedBrands(clean).length > 0)
      return `name: "${clean}" uses a brand's name; tokens are public and permanent, so pick another name.`;
    return null;
  } catch (err) {
    if (err instanceof UntrustedTextError) return err.message;
    throw err;
  }
}

/** Why `symbol` can't be a ticker, or null. Pure; exported for tests. */
export function tickerProblem(symbol: string): string | null {
  if (!/^[A-Z0-9]{3,6}$/.test(symbol)) return 'symbol: 3 to 6 capital letters or digits.';
  if ((TICKER_RESERVED as readonly string[]).includes(symbol) || reservedBrands(symbol).length > 0)
    return `symbol: ${symbol} reads as a brand's or a major asset's ticker; pick another.`;
  return null;
}

/** Ticker candidates in order (v2.3 §2). Pure; exported for tests. */
export function tickerCandidates(name: string): string[] {
  const words = name.toUpperCase().split(/[^A-Z0-9]+/).filter(Boolean);
  if (words.length === 0) return [];
  const out: string[] = [];
  if (words.length >= 3) out.push(words.map((w) => w[0]).join('').slice(0, 6));
  const first = words[0].length >= 3 ? words[0] : words.join('');
  if (first.length >= 3) {
    out.push(first.slice(0, 6));
    for (let d = 2; d <= 9; d++) out.push(`${first.slice(0, 5)}${d}`);
  }
  return [...new Set(out)].filter((c) => /^[A-Z0-9]{3,6}$/.test(c));
}

export type TickerCheck = (symbol: string) => Promise<boolean | null>;

/** tokens.fun's check-ticker; null when it can't be asked (no key, timeout, error). */
export const tokensFunTickerCheck: TickerCheck = async (symbol) => {
  const key = process.env.TOKENSFUN_DEPLOY_API_KEY;
  if (!key) return null;
  try {
    const res = await fetch(`${TOKENSFUN_URL}/api/check-ticker?symbol=${encodeURIComponent(symbol)}`, {
      headers: { Authorization: `Bearer ${key}` },
      signal: AbortSignal.timeout(3_000),
    });
    if (!res.ok) return null;
    const body = (await res.json()) as { available?: unknown };
    return typeof body.available === 'boolean' ? body.available : null;
  } catch {
    return null;
  }
};

export type Proposal =
  | { name: string; symbol: string; note?: string }
  | { name: null; symbol: null; reason: string };

/** EarnKit's proposed token for a project (v2.3 §2). */
export async function proposeToken(projectName: string, check: TickerCheck = tokensFunTickerCheck): Promise<Proposal> {
  const trimmed = projectName.trim().slice(0, 50);
  const name = !tokenNameProblem(trimmed)
    ? trimmed
    : !tokenNameProblem(`${trimmed.slice(0, 44)} Token`)
      ? `${trimmed.slice(0, 44)} Token`
      : null;
  const candidates = tickerCandidates(projectName).filter((c) => !tickerProblem(c));
  if (!name || candidates.length === 0)
    return { name: null, symbol: null, reason: "EarnKit couldn't propose a token for this name: pick a name and ticker with the builder." };
  for (const c of candidates) {
    const free = await check(c);
    if (free === null) return { name, symbol: candidates[0], note: 'ticker not checked for duplicates' };
    if (free) return { name, symbol: c };
  }
  return { name: null, symbol: null, reason: 'Every proposed ticker is taken: pick a name and ticker with the builder.' };
}
